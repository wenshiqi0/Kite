cmake_minimum_required(VERSION 3.20)
project(kite_windows_media LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(opus
  URL https://downloads.xiph.org/releases/opus/opus-1.5.2.tar.gz
  URL_HASH SHA256=65c1d2f78b9f2fb20082c38cbe47c951ad5839345876e46941612ee87f9a7ce1
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(opus)
add_library(kite_windows_media SHARED media.cpp)
target_compile_features(kite_windows_media PRIVATE cxx_std_20)
target_compile_definitions(kite_windows_media PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(kite_windows_media PRIVATE opus mfplat mfuuid ole32 winmm)
if(MINGW)
  target_link_options(kite_windows_media PRIVATE -static -static-libgcc -static-libstdc++)
  set_target_properties(kite_windows_media PROPERTIES PREFIX "")
endif()
add_executable(kite_media_test media_test.cpp)
target_compile_features(kite_media_test PRIVATE cxx_std_20)
target_link_libraries(kite_media_test PRIVATE kite_windows_media)
if(MINGW)
  target_link_options(kite_media_test PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_test(NAME decode COMMAND kite_media_test
  "${CMAKE_CURRENT_SOURCE_DIR}/../../macos/Tests/Fixtures/frame.h264")
